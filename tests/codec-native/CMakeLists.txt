cmake_minimum_required(VERSION 3.28)
project(retro_codec_native LANGUAGES C CXX)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_LIST_DIR}/../../cmake/common")
include(RetroFFmpeg)
enable_testing()
set(codec_src "${CMAKE_CURRENT_LIST_DIR}/../../src/codec/video-roundtrip.c")
foreach(test_name IN ITEMS h263-roundtrip video-codecs)
  add_executable("test-${test_name}" "../test-${test_name}.c" "${codec_src}")
  target_link_libraries("test-${test_name}" PRIVATE retro_ffmpeg)
  add_test(NAME "${test_name}" COMMAND "test-${test_name}")
endforeach()

add_executable(test-codec-worker "../test-codec-worker.cpp" "${codec_src}"
  "${CMAKE_CURRENT_LIST_DIR}/../../src/codec/codec-worker.cpp")
target_link_libraries(test-codec-worker PRIVATE retro_ffmpeg Threads::Threads)
add_test(NAME codec-worker COMMAND test-codec-worker)
